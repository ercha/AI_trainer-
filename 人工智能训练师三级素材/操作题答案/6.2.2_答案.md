# 6.2.2 模块B-提升：客户流失随机森林分类与参数优化 - 参考答案

```python
import pandas as pd,joblib
from sklearn.model_selection import train_test_split,GridSearchCV
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report,confusion_matrix
df=pd.read_csv("churn.csv")
# TODO 1-1：独热编码并拆分X/y
X=pd.get_dummies(df.drop(columns=["id","churn"]),drop_first=True);y=df["churn"]
# TODO 1-2：8:2分层划分
X_train,X_test,y_train,y_test=train_test_split(X,y,test_size=.2,random_state=42,stratify=y)
# TODO 2-1：基础随机森林
base=RandomForestClassifier(n_estimators=200,random_state=42,class_weight="balanced");base.fit(X_train,y_train);pred=base.predict(X_test);print(classification_report(y_test,pred));print(confusion_matrix(y_test,pred))
# TODO 2-2：参数网格
params={"n_estimators":[100,200],"max_depth":[None,6,10],"min_samples_leaf":[1,3]}
# TODO 2-3：GridSearchCV以f1为评分
grid=GridSearchCV(RandomForestClassifier(random_state=42,class_weight="balanced"),params,cv=3,scoring="f1",n_jobs=-1);grid.fit(X_train,y_train);print(grid.best_params_)
```
