# 6.2.1 模块B-基础：客户画像K-Means聚类与最佳K选择 - 参考答案

```python
import pandas as pd,matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
from sklearn.decomposition import PCA
df=pd.read_csv("customers.csv");features=["age","monthly_income","visit_frequency","monthly_spend","discount_sensitivity"]
# TODO 1-1：标准化
X=StandardScaler().fit_transform(df[features]);scores=[]
for k in range(2,7):
    # TODO 1-2：训练KMeans
    model=KMeans(n_clusters=k,random_state=42,n_init=10);labels=model.fit_predict(X)
    # TODO 1-3：轮廓系数
    scores.append((k,silhouette_score(X,labels)))
# TODO 2-1：选择最佳K
best_k=max(scores,key=lambda z:z[1])[0];model=KMeans(n_clusters=best_k,random_state=42,n_init=10);df["cluster"]=model.fit_predict(X)
# TODO 2-2：PCA降到二维
xy=PCA(n_components=2,random_state=42).fit_transform(X);df["pca1"],df["pca2"]=xy[:,0],xy[:,1];df.to_csv("customer_clusters.csv",index=False,encoding="utf-8-sig");plt.scatter(df.pca1,df.pca2,c=df.cluster);plt.savefig("clusters.png",dpi=150);print(scores)
```
