# 6.5.2 模块E-提升：客户投诉分析与结构化AI处置建议 - 参考答案

```python
import os,json,time,requests,pandas as pd
df=pd.read_csv("complaints.csv")
# TODO 1-1：类别×优先级统计
summary=df.groupby(["category","priority"]).size().unstack(fill_value=0);sample=df.groupby("category").head(3).to_dict("records");prompt=f"""你是客服分析助手。输出严格JSON对象，字段必须为summary、top_risks、actions、sla_advice。\n统计：{summary.to_dict()}\n样例：{sample}"""
base=os.getenv("OPENAI_BASE_URL","").rstrip("/");key=os.getenv("OPENAI_API_KEY","");model=os.getenv("OPENAI_MODEL","gpt-5-mini");result=None
if base and key:
    # TODO 2-1：最多重试3次，并发送response_format=json_object
    for attempt in range(3):
        try:
            resp=requests.post(base+"/v1/chat/completions",headers={"Authorization":"Bearer "+key,"Content-Type":"application/json"},json={"model":model,"messages":[{"role":"user","content":prompt}],"response_format":{"type":"json_object"}},timeout=60);resp.raise_for_status();text=resp.json()["choices"][0]["message"]["content"]
            # TODO 2-2：解析JSON，成功后退出重试
            result=json.loads(text);break
        except Exception as e:print("retry",attempt+1,e);time.sleep(1)
if result is None:result={"summary":summary.to_dict(),"top_risks":[],"actions":["配置API后重新生成"],"sla_advice":"离线训练模式"}
# TODO 3-1：保存UTF-8中文JSON
json.dump(result,open("投诉智能分析.json","w",encoding="utf-8"),ensure_ascii=False,indent=2);print(result)
```
