# 6.5.3 模块E-综合：本地知识库检索增强问答RAG - 参考答案

```python
import os,json,requests
from pathlib import Path
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
# TODO 1-1：读取knowledge目录
docs=[{"file":p.name,"text":p.read_text(encoding="utf-8")} for p in sorted(Path("knowledge").glob("*.txt"))];questions=json.load(open("questions.json",encoding="utf-8"))
# TODO 1-2：字符级TF-IDF
vec=TfidfVectorizer(analyzer="char",ngram_range=(2,4),min_df=1);matrix=vec.fit_transform([d["text"] for d in docs]);base=os.getenv("OPENAI_BASE_URL","").rstrip("/");key=os.getenv("OPENAI_API_KEY","");model=os.getenv("OPENAI_MODEL","gpt-5-mini");results=[]
for item in questions:
    q=item["question"]
    # TODO 2-1：余弦相似度取Top2证据
    qv=vec.transform([q]);scores=cosine_similarity(qv,matrix)[0];top=scores.argsort()[::-1][:2];evidence=[docs[i] for i in top];context="\n".join(x["text"] for x in evidence);prompt=f"仅根据以下制度回答，证据不足请明确说明。\n制度：{context}\n问题：{q}"
    if base and key:
        # TODO 2-2：调用OpenAI兼容接口
        resp=requests.post(base+"/v1/chat/completions",headers={"Authorization":"Bearer "+key,"Content-Type":"application/json"},json={"model":model,"messages":[{"role":"user","content":prompt}]},timeout=60);resp.raise_for_status();answer=resp.json()["choices"][0]["message"]["content"]
    else:answer="[离线模式] "+context
    results.append({"question":q,"evidence":[x["file"] for x in evidence],"scores":[float(scores[i]) for i in top],"answer":answer})
# TODO 3-1：保存带证据链结果
json.dump(results,open("rag_result.json","w",encoding="utf-8"),ensure_ascii=False,indent=2);print(results)
```
