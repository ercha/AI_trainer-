# 6.1.3 模块A-综合：交通图像质量筛选与类别平衡 - 参考答案

```python
from pathlib import Path
from collections import Counter
import csv,random,cv2,pandas as pd
from PIL import Image,ImageEnhance
src=cv2.imread("traffic_grid.ppm");lab=pd.read_csv("traffic_labels.csv");OUT=Path("balanced");OUT.mkdir(exist_ok=True);valid=[]
# TODO 1-1：按5列×2行切分10张48×32图
images=[src[r*32:(r+1)*32,c*48:(c+1)*48].copy() for r in range(2) for c in range(5)]
for i,img in enumerate(images):
    # TODO 1-2：拉普拉斯方差清晰度
    sharp=cv2.Laplacian(cv2.cvtColor(img,cv2.COLOR_BGR2GRAY),cv2.CV_64F).var()
    if sharp>=5:valid.append((i,img,lab.iloc[i]["class"],sharp))
counts=Counter(v[2] for v in valid)
# TODO 2-1：目标样本数取最大类别
n=max(counts.values());rows=[];random.seed(42)
for cls in sorted(counts):
    items=[v for v in valid if v[2]==cls];d=OUT/cls;d.mkdir(parents=True,exist_ok=True)
    for i,img,_,s in items:cv2.imwrite(str(d/f"orig_{i}.png"),img);rows.append([cls,"orig",s])
    while len([r for r in rows if r[0]==cls])<n:
        i,img,_,s=random.choice(items);pil=Image.fromarray(cv2.cvtColor(img,cv2.COLOR_BGR2RGB))
        # TODO 2-2：旋转并调节亮度
        aug=ImageEnhance.Brightness(pil.rotate(random.choice([-10,-5,5,10]))).enhance(random.uniform(.85,1.15));idx=len([r for r in rows if r[0]==cls]);aug.save(d/f"aug_{idx}.png");rows.append([cls,"aug",s])
# TODO 3-1：保存清单
with open("balanced_manifest.csv","w",encoding="utf-8-sig",newline="") as f:
    w=csv.writer(f);w.writerow(["class","source","sharpness"]);w.writerows(rows)
```
