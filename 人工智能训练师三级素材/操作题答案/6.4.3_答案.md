# 6.4.3 模块D-综合：LDA主题建模与相似文档推荐 - 参考答案

```python
import pandas as pd,numpy as np
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.decomposition import LatentDirichletAllocation
from sklearn.metrics.pairwise import cosine_similarity
df=pd.read_csv("documents.csv")
# TODO 1-1：CountVectorizer，最多500特征
vec=CountVectorizer(max_features=500,min_df=2);X=vec.fit_transform(df.text)
# TODO 1-2：训练5主题LDA
lda=LatentDirichletAllocation(n_components=5,random_state=42,learning_method="batch");doc_topic=lda.fit_transform(X);terms=np.array(vec.get_feature_names_out())
for i,comp in enumerate(lda.components_):
    # TODO 2-1：每主题Top6关键词
    idx=np.argsort(comp)[-6:][::-1];print(i,terms[idx].tolist())
# TODO 2-2：计算主题分布余弦相似度
sim=cosine_similarity(doc_topic);rows=[]
for i in range(len(df)):
    # TODO 2-3：排除自身，取Top3相似文档
    top=np.argsort(sim[i])[::-1][1:4]
    for j in top:rows.append([int(df.iloc[i].id),int(df.iloc[j].id),float(sim[i,j])])
pd.DataFrame(rows,columns=["doc_id","recommend_id","similarity"]).to_csv("recommendations.csv",index=False)
```
