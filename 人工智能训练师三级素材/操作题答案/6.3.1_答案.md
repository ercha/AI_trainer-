# 6.3.1 模块C-基础：CNN图形图像四分类 - 参考答案

```python
import cv2,pandas as pd,torch
from torch import nn,optim
from torch.utils.data import TensorDataset,DataLoader,random_split
img=cv2.imread("shapes_grid.pgm",0);labels=pd.read_csv("shape_labels.csv").label.values;tiles=[]
# TODO 1-1：切分8×20个16×16图块
for r in range(8):
    for c in range(20):tiles.append(img[r*16:(r+1)*16,c*16:(c+1)*16])
# TODO 1-2：转为[N,1,16,16]浮点Tensor
X=torch.tensor(tiles,dtype=torch.float32).unsqueeze(1)/255.0;y=torch.tensor(labels,dtype=torch.long);ds=TensorDataset(X,y);n=int(len(ds)*.8);train_ds,test_ds=random_split(ds,[n,len(ds)-n],generator=torch.Generator().manual_seed(42));train=DataLoader(train_ds,32,shuffle=True);test=DataLoader(test_ds,64)
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        # TODO 2-1：两层Conv+ReLU+MaxPool
        self.features=nn.Sequential(nn.Conv2d(1,16,3,padding=1),nn.ReLU(),nn.MaxPool2d(2),nn.Conv2d(16,32,3,padding=1),nn.ReLU(),nn.MaxPool2d(2));self.fc=nn.Linear(32*4*4,4)
    def forward(self,x):return self.fc(self.features(x).flatten(1))
model=Net()
# TODO 2-2：交叉熵和Adam
loss_fn=nn.CrossEntropyLoss();optimizer=optim.Adam(model.parameters(),lr=1e-3)
for _ in range(5):
    for x,b in train:optimizer.zero_grad();loss=loss_fn(model(x),b);loss.backward();optimizer.step()
correct=total=0;model.eval()
with torch.no_grad():
    for x,b in test:
        # TODO 3-1：预测类别
        pred=model(x).argmax(1);correct+=(pred==b).sum().item();total+=b.numel()
print("accuracy=",correct/total);torch.save(model.state_dict(),"cnn_shapes.pth")
```
