# 6.4.2 模块D-提升：Word2Vec影评向量与情感分类 - 参考答案

```python
import pandas as pd,numpy as np
from gensim.models import Word2Vec
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import classification_report
df=pd.read_csv("reviews.csv");sentences=[t.split() for t in df.text]
# TODO 1-1：训练100维Word2Vec
w2v=Word2Vec(sentences,vector_size=100,window=5,min_count=1,workers=1,seed=42);w2v.save("word2vec.model")
def sent_vec(text):
    # TODO 1-2：有效词向量取平均
    vecs=[w2v.wv[w] for w in text.split() if w in w2v.wv];return np.mean(vecs,axis=0) if vecs else np.zeros(w2v.vector_size)
X=np.vstack([sent_vec(t) for t in df.text]);y=df.label.values;Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.2,random_state=42,stratify=y)
# TODO 2-1：LogisticRegression分类
model=LogisticRegression(max_iter=1000,random_state=42);model.fit(Xtr,ytr);pred=model.predict(Xte);print(classification_report(yte,pred,zero_division=0))
# TODO 2-2：查询“精彩”Top5相似词
print(w2v.wv.most_similar("精彩",topn=5))
```
