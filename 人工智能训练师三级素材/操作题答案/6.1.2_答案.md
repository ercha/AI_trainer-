# 6.1.2 模块A-提升：工业零件缺陷分割与统计 - 参考答案

```python
from pathlib import Path
import csv,cv2,numpy as np
src=cv2.imread("defects_grid.ppm");OUT=Path("defect_output");OUT.mkdir(exist_ok=True);rows=[]
# TODO 1-1：按4列×2行切分8张64×64图
images=[src[r*64:(r+1)*64,c*64:(c+1)*64].copy() for r in range(2) for c in range(4)]
for i,img in enumerate(images):
    # TODO 1-2：灰度化
    gray=cv2.cvtColor(img,cv2.COLOR_BGR2GRAY);blur=cv2.GaussianBlur(gray,(5,5),0)
    # TODO 1-3：阈值分割亮缺陷
    _,mask=cv2.threshold(blur,160,255,cv2.THRESH_BINARY);k=cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(5,5))
    # TODO 1-4：形态学开运算
    clean=cv2.morphologyEx(mask,cv2.MORPH_OPEN,k)
    # TODO 1-5：外轮廓
    contours,_=cv2.findContours(clean,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE);area=sum(cv2.contourArea(c) for c in contours);rows.append([i,len(contours),area])
    out=img.copy();cv2.drawContours(out,contours,-1,(0,0,255),2);cv2.imwrite(str(OUT/f"part_{i}.png"),out)
# TODO 2-1：保存缺陷报告
with open("defect_report.csv","w",encoding="utf-8-sig",newline="") as f:
    w=csv.writer(f);w.writerow(["index","count","area"]);w.writerows(rows)
```
