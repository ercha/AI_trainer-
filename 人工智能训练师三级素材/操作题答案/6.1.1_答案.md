# 6.1.1 模块A-基础：商品图片质量检查与离线增强 - 参考答案

```python
from pathlib import Path
import csv
from PIL import Image,ImageEnhance,ImageOps
SRC=Image.open("product_grid.ppm").convert("RGB");OUT=Path("augmented");OUT.mkdir(exist_ok=True)
# TODO 1-1：按3列×2行切分6张32×32图片
images=[SRC.crop((c*32,r*32,(c+1)*32,(r+1)*32)) for r in range(2) for c in range(3)]
manifest=[]
for i,img in enumerate(images):
    # TODO 2-1：统一为128×128
    img=img.resize((128,128))
    img.save(OUT/f"product_{i}.png");manifest.append([f"product_{i}.png","original"])
    # TODO 2-2：水平翻转
    a=ImageOps.mirror(img);a.save(OUT/f"product_{i}_flip.png");manifest.append([f"product_{i}_flip.png","flip"])
    # TODO 2-3：旋转12度
    b=img.rotate(12);b.save(OUT/f"product_{i}_rot.png");manifest.append([f"product_{i}_rot.png","rotate"])
    # TODO 2-4：亮度增强1.2倍
    c=ImageEnhance.Brightness(img).enhance(1.2);c.save(OUT/f"product_{i}_bright.png");manifest.append([f"product_{i}_bright.png","brightness"])
# TODO 3-1：保存清单
with open("manifest.csv","w",encoding="utf-8-sig",newline="") as f:
    w=csv.writer(f);w.writerow(["file","type"]);w.writerows(manifest)
```
