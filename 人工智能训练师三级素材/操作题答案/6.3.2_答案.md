```python
import cv2,pandas as pd,torch,random
from torch import nn,optim
from torch.utils.data import Dataset,DataLoader
from sklearn.metrics import confusion_matrix
img=cv2.imread("shapes_grid.pgm",0);labels=pd.read_csv("shape_labels.csv").label.values;tiles=[img[r*16:(r+1)*16,c*16:(c+1)*16] for r in range(8) for c in range(20)]
class DS(Dataset):
    def __init__(self,ids,aug=False):self.ids=ids;self.aug=aug
    def __len__(self):return len(self.ids)
    def __getitem__(self,i):
        k=self.ids[i];x=torch.tensor(tiles[k],dtype=torch.float32).unsqueeze(0)/255
        # TODO 1-1：随机水平翻转/90度旋转
        if self.aug:\n            if random.random()<.5:x=torch.flip(x,[2])\n            if random.random()<.5:x=torch.rot90(x,random.choice([1,3]),[1,2])
        return x,int(labels[k])
ids=list(range(160));random.Random(42).shuffle(ids);train=DataLoader(DS(ids[:112],True),32,shuffle=True);val=DataLoader(DS(ids[112:136]),64);test=DataLoader(DS(ids[136:]),64)
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        # TODO 1-2：BatchNorm+Dropout网络
        self.net=nn.Sequential(nn.Conv2d(1,16,3,padding=1),nn.BatchNorm2d(16),nn.ReLU(),nn.MaxPool2d(2),nn.Conv2d(16,32,3,padding=1),nn.BatchNorm2d(32),nn.ReLU(),nn.MaxPool2d(2),nn.Flatten(),nn.Dropout(.35),nn.Linear(32*4*4,4))
    def forward(self,x):return self.net(x)
model=Net();loss_fn=nn.CrossEntropyLoss();opt=optim.Adam(model.parameters(),lr=1e-3)
# TODO 2-1：StepLR每3轮×0.5
scheduler=optim.lr_scheduler.StepLR(opt,step_size=3,gamma=.5);best=0
for _ in range(8):
    model.train()
    for x,b in train:opt.zero_grad();loss=loss_fn(model(x),b);loss.backward();opt.step()
    scheduler.step();model.eval();c=n=0
    with torch.no_grad():
        for x,b in val:p=model(x).argmax(1);c+=(p==b).sum().item();n+=b.numel()
    acc=c/n
    # TODO 2-2：保存最佳模型
    if acc>best:best=acc;torch.save(model.state_dict(),"best_model.pth")
model.load_state_dict(torch.load("best_model.pth",weights_only=True));ys=[];ps=[]
with torch.no_grad():
    for x,b in test:ps+=model(x).argmax(1).tolist();ys+=b.tolist()
# TODO 3-1：输出混淆矩阵
print(confusion_matrix(ys,ps))
```
