# 6.5.1 模块E-基础：电商经营分析与AI经营报告 - 参考答案

```python
import os,requests,pandas as pd
import matplotlib.pyplot as plt
df=pd.read_csv("ecommerce.csv")
# TODO 1-1：新增profit与return_rate
df["profit"]=df["revenue"]-df["cost"];df["return_rate"]=df["returns"]/df["orders"].clip(lower=1)
# TODO 1-2：按category汇总经营指标
summary=df.groupby("category").agg(revenue=("revenue","sum"),profit=("profit","sum"),orders=("orders","sum"),returns=("returns","sum"),rating=("rating","mean"));print(summary);summary[["revenue","profit"]].plot(kind="bar");plt.tight_layout();plt.savefig("经营指标.png",dpi=160)
# TODO 2-1：构造包含“数据解读、经营问题、商业结论、优化建议”的Prompt
prompt=f"请根据以下经营汇总生成Markdown报告，必须包含：数据解读、经营问题、商业结论、优化建议。\\n{summary.to_string()}";base=os.getenv("OPENAI_BASE_URL","").rstrip("/");key=os.getenv("OPENAI_API_KEY","");model=os.getenv("OPENAI_MODEL","gpt-5-mini")
if base and key:
    # TODO 2-2：调用OpenAI兼容接口
    resp=requests.post(base+"/v1/chat/completions",headers={"Authorization":"Bearer "+key,"Content-Type":"application/json"},json={"model":model,"messages":[{"role":"user","content":prompt}]},timeout=60);resp.raise_for_status();report=resp.json()["choices"][0]["message"]["content"]
else:report="# 离线训练\n\n未配置API。\n\n"+prompt
open("经营分析报告.md","w",encoding="utf-8").write(report);print(report[:800])
```
