# 6.4.1 模块D-基础：TF-IDF新闻分类与关键词分析 - 参考答案

```python
import pandas as pd,numpy as np
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.svm import LinearSVC
from sklearn.metrics import classification_report,f1_score
df=pd.read_csv("news.csv");X_train,X_test,y_train,y_test=train_test_split(df.text,df.label,test_size=.2,random_state=42,stratify=df.label)
# TODO 1-1：TF-IDF 1~2 gram，最多1200特征
vec=TfidfVectorizer(ngram_range=(1,2),max_features=1200,min_df=2);A=vec.fit_transform(X_train);B=vec.transform(X_test)
# TODO 1-2：训练LinearSVC
model=LinearSVC(random_state=42);model.fit(A,y_train);pred=model.predict(B);print(classification_report(y_test,pred));print("macro_f1=",f1_score(y_test,pred,average="macro"))
terms=np.array(vec.get_feature_names_out())
for i,cls in enumerate(model.classes_):
    # TODO 2-1：提取每类Top8关键词
    idx=np.argsort(model.coef_[i])[-8:][::-1];print(cls,terms[idx].tolist())
```
